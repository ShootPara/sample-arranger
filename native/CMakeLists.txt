cmake_minimum_required(VERSION 3.25)
project(SampleArrangerNative LANGUAGES C CXX)
if(NOT WIN32 OR NOT VST3_SDK_ROOT)
    message(FATAL_ERROR "Supply the pinned Windows VST3_SDK_ROOT")
endif()
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "")
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "")
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "")
add_subdirectory("${VST3_SDK_ROOT}" sdk)
set(hosting "${VST3_SDK_ROOT}/public.sdk/source/vst/hosting")
add_executable(arranger-vst3 worker.cpp "${hosting}/module_win32.cpp" "${hosting}/plugprovider.cpp"
    "${VST3_SDK_ROOT}/public.sdk/source/vst/vstpresetfile.cpp"
    "${VST3_SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
set_target_properties(arranger-vst3 PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
target_link_libraries(arranger-vst3 PRIVATE sdk_hosting)
if(MINGW)
    target_link_options(arranger-vst3 PRIVATE -municode)
endif()
add_library(arranger-delay-fixture MODULE delay_fixture.cpp
    "${VST3_SDK_ROOT}/public.sdk/source/main/dllmain.cpp")
target_link_libraries(arranger-delay-fixture PRIVATE sdk)
set_target_properties(arranger-delay-fixture PROPERTIES PREFIX "" SUFFIX ".vst3"
    CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
add_executable(arranger-playback playback.cpp)
target_link_libraries(arranger-playback PRIVATE winmm)
set_target_properties(arranger-playback PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
add_executable(arranger-loopback-probe loopback_probe.cpp)
target_link_libraries(arranger-loopback-probe PRIVATE mmdevapi ole32 uuid)
set_target_properties(arranger-loopback-probe PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
add_executable(arranger-lifecycle-fixture lifecycle_fixture.cpp)
set_target_properties(arranger-lifecycle-fixture PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
if(MINGW)
    target_link_options(arranger-lifecycle-fixture PRIVATE -municode)
endif()
